cmake_minimum_required(VERSION 3.20)
project(curvatura CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(curvatura INTERFACE)
target_include_directories(curvatura INTERFACE include)
add_executable(curvatura-field src/main.cpp)
target_link_libraries(curvatura-field curvatura)
enable_testing()
add_executable(test_alcubierre tests/test_alcubierre.cpp)
target_link_libraries(test_alcubierre curvatura)
add_test(NAME alcubierre COMMAND test_alcubierre)
